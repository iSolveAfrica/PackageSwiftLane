const res = await fetch("https://swiftlane.gh/api/v1/shipments?status=in_transit", { headers: { Authorization: `Bearer ${key}` } }); const { data, meta } = await res.json();
$ch = curl_init("https://swiftlane.gh/api/v1/packages"); curl_setopt($ch, CURLOPT_HTTPHEADER, ["Authorization: Bearer $key"]); $packages = json_decode(curl_exec($ch), true);

Every response is JSON: data, meta for paging (page, per_page, sort) and error on failure.
